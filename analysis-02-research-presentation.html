WITH t AS (
  SELECT o.order_id, SUM(p.unit_price * od.quantity * (1 - od.discount)) AS x, o.freight AS y
  FROM orders AS o
  JOIN order_details AS od ON od.order_id = o.order_id
  JOIN products AS p ON p.product_id = od.product_id
  GROUP BY o.order_id
)
SELECT COUNT(*) AS n,
       AVG(x) AS mean_x, AVG(y) AS mean_y,
       (AVG(x * y) - AVG(x) * AVG(y))
         / (AVG(x * x) - AVG(x) * AVG(x)) AS slope,
       AVG(y) - (AVG(x * y) - AVG(x) * AVG(y))
         / (AVG(x * x) - AVG(x) * AVG(x)) * AVG(x) AS intercept,
       (AVG(x * y) - AVG(x) * AVG(y))
         / SQRT((AVG(x * x) - AVG(x) * AVG(x))
              * (AVG(y * y) - AVG(y) * AVG(y))) AS r
FROM t;

       COUNT(DISTINCT e.employee_id) AS emp,

SELECT s.company_name,
       COUNT(*) AS shipped,
       SUM(o.shipped_date > o.required_date) AS late,
       ROUND(100.0 * AVG(o.shipped_date > o.required_date), 1) AS pct_late,
       ROUND(AVG(CASE WHEN o.shipped_date > o.required_date
             THEN julianday(o.shipped_date) - julianday(o.required_date) END), 1) AS days_late
FROM orders AS o
JOIN shippers AS s ON s.shipper_id = o.ship_via
WHERE o.shipped_date IS NOT NULL
GROUP BY s.company_name
ORDER BY pct_late;

WITH t AS (
  SELECT (YEAR(o.order_date) - 2016) * 12 + MONTH(o.order_date) - 6 AS x,
         SUM(p.unit_price * od.quantity * (1 - od.discount)) AS y
  FROM orders AS o
  JOIN order_details AS od ON od.order_id = o.order_id
  JOIN products AS p ON p.product_id = od.product_id
  WHERE o.order_date < '2018-05-01'
  GROUP BY x
)
SELECT COUNT(*) AS n,
       AVG(x) AS mean_x, AVG(y) AS mean_y,
       (AVG(x * y) - AVG(x) * AVG(y))
         / (AVG(x * x) - AVG(x) * AVG(x)) AS slope,
       AVG(y) - (AVG(x * y) - AVG(x) * AVG(y))
         / (AVG(x * x) - AVG(x) * AVG(x)) * AVG(x) AS intercept,
       (AVG(x * y) - AVG(x) * AVG(y))
         / SQRT((AVG(x * x) - AVG(x) * AVG(x))
              * (AVG(y * y) - AVG(y) * AVG(y))) AS r
FROM t;

  SELECT o.customer_id, SUM(p.unit_price * od.quantity * (1 - od.discount)) AS sales
  FROM orders AS o

r AS (SELECT sales, ROW_NUMBER() OVER (ORDER BY sales DESC) AS rn FROM t)
SELECT COUNT(*) AS customers,

       ROUND(100.0 * SUM(CASE WHEN rn <= 30 THEN sales END) / SUM(sales), 1) AS top30
FROM r;
